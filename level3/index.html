หมวด 1: #include <cmath> (กระเป๋าคำนวณคณิตศาสตร์)

หน้าที่หลัก: คำนวณฟังก์ชันทางคณิตศาสตร์ระดับสูง เช่น ถอดรากที่สอง (Square Root), ยกกำลัง, ปัดเศษทศนิยม และหาค่าสัมบูรณ์

MATH_ADVANCED.CPP C++ 
#include <iostream>
#include <cmath> // 1. หยิบกระเป๋าคำนวณคณิตศาสตร์ขั้นสูงมาใช้งาน

using namespace std;

int main() {
    double root_val = sqrt(16); // 2. ถอดสแควร์รูทของ 16 (ผลลัพธ์คือ 4.0)
    double pow_val = pow(2, 3);   // 3. คำนวณ 2 ยกกำลัง 3 (2*2*2 = 8.0)
    int abs_val = abs(-15);       // 4. หาค่าสัมบูรณ์ ถอดเครื่องหมายลบทิ้ง (ผลลัพธ์คือ 15)
    
    cout << "Sqrt: " << root_val << ", Pow: " << pow_val << ", Abs: " << abs_val << "\n";
    cout << "Ceil 3.2: " << ceil(3.2) << "\n"; // 5. ปัดเศษขึ้นเสมอ (ผลลัพธ์คือ 4)
    cout << "Floor 3.8: " << floor(3.8) << "\n"; // 6. ปัดเศษลงเสมอ (ผลลัพธ์คือ 3)
    
    return 0;
}

หมวด 2: #include <queue> & <stack> (กระเป๋าคิวและกองซ้อน)

หน้าที่หลัก: Queue คือ "เข้าก่อน-ออกก่อน" (FIFO) เหมือนต่อคิวซื้อของ | Stack คือ "เข้าหลังสุด-ออกก่อน" (LIFO) เหมือนจานซ้อนกัน

QUEUE_STACK_DEMO.CPP C++ 
#include <iostream>
#include <queue> // 1. หยิบกระเป๋า Queue (เข้าก่อน ออกก่อน)
#include <stack> // 2. หยิบกระเป๋า Stack (เข้าทีหลัง ออกก่อน)

using namespace std;

int main() {
    // --- ระบบ Queue (ต่อคิว) ---
    queue<int> q;
    q.push(10); // คิวที่ 1
    q.push(20); // คิวที่ 2
    cout << "First in Queue: " << q.front() << "\n"; // ดูคนหัวแถว (ผลลัพธ์คือ 10)
    q.pop(); // เอาคนหัวแถวออกไป

    // --- ระบบ Stack (กองซ้อน) ---
    stack<int> s;
    s.push(100); // วางจานใบที่ 1
    s.push(200); // วางจานใบที่ 2 ซ้อนทับ
    cout << "Top of Stack: " << s.top() << "\n"; // ดูจานใบบนสุด (ผลลัพธ์คือ 200)
    s.pop(); // หยิบจานใบบนสุดออก
    
    return 0;
}

หมวด 3: 2D Grid (กระเป๋าตาราง 2 มิติ / เมทริกซ์)

หน้าที่หลัก: เก็บข้อมูลเป็นตารางมีแถว (Row) และคอลัมน์ (Column) นิยมใช้ทำโจทย์ตารางพิกัด แผนที่เขาวงกต และเมทริกซ์

GRID_2D_DEMO.CPP C++ 
#include <iostream>
#include <vector>

using namespace std;

int main() {
    // 1. สร้างตารางขนาด 2 แถว 3 คอลัมน์
    vector<vector<int>> grid = {
        {1, 2, 3}, // แถวที่ 0
        {4, 5, 6}  // แถวที่ 1
    };
    
    // 2. เข้าถึงข้อมูลแถวที่ 1 คอลัมน์ที่ 2 (นับเริ่มจาก 0)
    cout << "Value at row 1, col 2: " << grid[1][2] << "\n"; // ผลลัพธ์คือ 6
    
    // 3. วนลูปอ่านค่าทั้งตาราง
    for (int r = 0; r < 2; r++) {
        for (int c = 0; c < 3; c++) {
            cout << grid[r][c] << " ";
        }
        cout << "\n";
    }
    
    return 0;
}
