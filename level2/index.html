หมวด 1: #include <iostream>

หน้าที่หลัก: ใช้รับข้อมูลเข้าจากผู้ใช้หรือคีย์บอร์ด และแสดงผลลัพธ์ออกทางหน้าจอ เปรียบเหมือนระบบช่องทางส่งเข้า-ออกหลักของบ้าน

MAIN_IOSTREAM.CPP C++ 
#include <iostream> // 1. หยิบกระเป๋าหลักสำหรับรับค่าและแสดงผลมาเตรียมไว้

using namespace std; // 2. ขอใช้พื้นที่มาตรฐานของ C++ เพื่อความสะดวกในการเรียกคำสั่ง

int main() { // 3. จุดเริ่มต้นการทำงานของโปรแกรม
    int score = 0; // 4. ประกาศตัวแปรเก็บคะแนน เริ่มต้นเท่ากับ 0
    
    cout << "Enter your score: "; // 5. แสดงข้อความสั่งให้ผู้ใช้กรอกข้อมูล
    cin >> score; // 6. รับค่าตัวเลขจากคีย์บอร์ดมาเก็บไว้ในตัวแปร score
    
    cout << "Your score is: " << score << "\n"; // 7. แสดงผลลัพธ์ออกทางหน้าจอพร้อมขึ้นบรรทัดใหม่
    
    return 0; // 8. จบการทำงานของโปรแกรมอย่างสมบูรณ์
}
📦

หมวด 2: #include <vector>

หน้าที่หลัก: จัดการกลุ่มข้อมูลประเภทเดียวกันหลายๆ ตัว สามารถเพิ่ม ลด หรือขยายขนาดกล่องได้ตามต้องการเหมือนกล่องพลาสติกอเนกประสงค์

VECTOR_DEMO.CPP C++ 
#include <iostream>
#include <vector>   // 1. หยิบกระเป๋า Vector สำหรับสร้างกล่องเก็บข้อมูลแบบยืดหยุ่น

using namespace std;

int main() {
    vector<int> scores; // 2. สร้างกล่อง Vector เก็บเลขจำนวนเต็ม (เริ่มต้นเป็นกล่องว่าง)
    
    scores.push_back(50); // 3. ยัดข้อมูลเลข 50 เข้าไปต่อท้ายกล่องเป็นช่องแรก
    scores.push_back(85); // 4. ยัดข้อมูลเลข 85 เข้าไปต่อท้ายกล่องเพิ่มอีกตัว
    
    cout << "Total items: " << scores.size() << "\n"; // 5. เช็กขนาดข้อมูลในกล่อง (ผลลัพธ์คือ 2)
    cout << "First item: " << scores[0] << "\n";     // 6. ดึงข้อมูลช่องแรก (ตำแหน่งที่ 0) ออกมาดู (ผลลัพธ์คือ 50)
    
    return 0;
}
⚙️

หมวด 3: #include <algorithm>

หน้าที่หลัก: จัดการเรียงลำดับชุดข้อมูล (เช่น เรียงน้อยไปมาก) หรือค้นหาค่าต่ำสุด-สูงสุดได้อย่างรวดเร็วในบรรทัดเดียว

ALGORITHM_DEMO.CPP C++ 
#include <iostream>
#include <vector>
#include <algorithm> // 1. หยิบกระเป๋า Algorithm สำหรับใช้คำสั่งเรียงลำดับและจัดการคำนวณ

using namespace std;

int main() {
    vector<int> numbers = {40, 10, 30, 20}; // 2. สร้างกล่อง Vector และใส่ตัวเลขสลับตำแหน่งไว้
    
    // 3. สั่งเรียงลำดับข้อมูลในกล่องจากน้อยไปมาก (ตั้งแต่ตัวแรก .begin() ถึงตัวสุดท้าย .end())
    sort(numbers.begin(), numbers.end()); 
    
    cout << "Sorted first item: " << numbers[0] << "\n"; // 4. แสดงผลข้อมูลช่องแรกหลังเรียงแล้ว (ผลลัพธ์คือ 10)
    cout << "Maximum value: " << max(15, 25) << "\n";    // 5. หาค่าที่มากที่สุดระหว่าง 15 กับ 25 (ผลลัพธ์คือ 25)
    
    return 0;
}

หมวด 4: #include <string>

หน้าที่หลัก: จัดการข้อความ ตัวอักษรยาวๆ หรือประโยคคำพูด รวมถึงนับจำนวนและความยาวของตัวอักษร

STRING_DEMO.CPP C++ 
#include <iostream>
#include <string> // 1. หยิบกระเป๋า String สำหรับจัดการข้อความโดยเฉพาะ

using namespace std;

int main() {
    string name = "DekMorton"; // 2. สร้างตัวแปรเก็บข้อความ
    
    cout << "Hello, " << name << "\n"; // 3. แสดงข้อความทักทายออกหน้าจอ
    cout << "Text length: " << name.length() << "\n"; // 4. นับความยาวตัวอักษร (ผลลัพธ์คือ 9)
    cout << "First letter: " << name[0] << "\n";     // 5. ดึงตัวอักษรตำแหน่งแรกสุดมาเช็ก (ผลลัพธ์คือ 'D')
    
    return 0;
}
